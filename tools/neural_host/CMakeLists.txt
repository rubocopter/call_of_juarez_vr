cmake_minimum_required(VERSION 3.25)
project(CoJNeuralHost LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "CoJNeuralHost requires Windows x64 (-A x64)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Offline, pinned header only. Never use an external source checkout as input.
set(ngx_default "${CMAKE_CURRENT_BINARY_DIR}/neural-sdk/ngx/nvsdk_ngx_defs.h")
if(NOT EXISTS "${ngx_default}")
    set(ngx_default "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/deps/nvidia-dlss/include/nvsdk_ngx_defs.h")
endif()
set(COJ_NEURAL_NGX_DEFS "${ngx_default}"
    CACHE FILEPATH "Local NVIDIA DLSS definitions header (pinned SHA-256)")
if(NOT EXISTS "${COJ_NEURAL_NGX_DEFS}")
    message(FATAL_ERROR "Supply the pinned header with -DCOJ_NEURAL_NGX_DEFS=<nvsdk_ngx_defs.h>; see README.md")
endif()
file(SHA256 "${COJ_NEURAL_NGX_DEFS}" ngx_hash)
if(NOT ngx_hash STREQUAL "ea23f33497cd274860d1c25a97644fce807dcb0037c594547203343103fad03e")
    message(FATAL_ERROR "NVIDIA definitions SHA-256 mismatch")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/neural-sdk/ngx")
configure_file("${COJ_NEURAL_NGX_DEFS}" "${CMAKE_CURRENT_BINARY_DIR}/neural-sdk/ngx/nvsdk_ngx_defs.h" COPYONLY)

add_executable(CoJNeuralHost CoJNeuralHost.cpp src/NeuralPass.cpp)
target_include_directories(CoJNeuralHost PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/neural-sdk"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../src/backends/openvr")
target_compile_definitions(CoJNeuralHost PRIVATE NOMINMAX= WIN32_LEAN_AND_MEAN=)
target_link_libraries(CoJNeuralHost PRIVATE d3d11 d3d12 dxgi d3dcompiler advapi32)
add_executable(CoJNeuralImageLab CoJNeuralImageLab.cpp src/NeuralPass.cpp)
target_include_directories(CoJNeuralImageLab PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/neural-sdk")
target_compile_definitions(CoJNeuralImageLab PRIVATE NOMINMAX= WIN32_LEAN_AND_MEAN=)
target_link_libraries(CoJNeuralImageLab PRIVATE d3d11 d3d12 dxgi d3dcompiler advapi32)
add_library(CoJNeuralForwarder SHARED vendor/nr-bridge/nr_forwarder.cpp)
set_target_properties(CoJNeuralForwarder PROPERTIES PREFIX "" OUTPUT_NAME "nvngx.dll_nr_bridge")

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME neural_image_lab_cli COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/image_lab.py" "$<TARGET_FILE:CoJNeuralImageLab>")
    set_tests_properties(neural_image_lab_cli PROPERTIES TIMEOUT 60)
    add_executable(cojvr_neural_host_smoke tests/smoke.cpp)
    target_include_directories(cojvr_neural_host_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/backends/openvr")
    target_compile_definitions(cojvr_neural_host_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_dependencies(cojvr_neural_host_smoke CoJNeuralHost)
    add_test(NAME neural_host_synthetic COMMAND cojvr_neural_host_smoke "$<TARGET_FILE:CoJNeuralHost>")
    set_tests_properties(neural_host_synthetic PROPERTIES TIMEOUT 30)
endif()
